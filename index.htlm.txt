<script>
    const gift = document.getElementById('gift');
    const lid = document.getElementById('lid');
    const message = document.getElementById('message');
    const textEl = document.getElementById('text');
    const music = document.getElementById('music');
    const container = document.getElementById('container');
    const sun = document.getElementById('sun');

    const poema = `Belu, ha pasado el tiempo y siempre estuve pensando en vos... 

No te mentía cuando dije que te amaba con todo el corazón. Por más que intente seguir adelante, siempre recordaba ese lugar seguro y tranquilo en donde está el hermoso árbol bugambilia que siempre imaginé a tu lado. 

Sabes que todas serán flores moradas que nunca me llegarán como vos y a la final me dio un miedo al futuro... creo que inventé el nombre de esta enfermedad ficticia "futurofobia".

Es el pánico a un mañana donde vos no estés presente, porque mi futuro no es futuro si no estás aquí. Sé que a veces tenemos que pegar volantazos, pero si conseguimos unir todos los pedazos, tendremos la obra más bonita en nuestro regazo... y la guardaremos para siempre con un solo abrazo. 💛`;

    function typeWriter(text, i = 0) {
        if (i < text.length) {
            textEl.innerHTML += text[i];
            message.scrollTop = message.scrollHeight; 
            setTimeout(() => typeWriter(text, i + 1), 35);
        }
    }

    function createPetal() {
        const p = document.createElement('div');
        p.className = 'petal';
        const flowers = ['💜', '🌸', '🌺', '🌼', '✨'];
        p.innerText = flowers[Math.floor(Math.random() * flowers.length)];
        p.style.left = Math.random() * 100 + "vw";
        p.style.fontSize = (Math.random() * 15 + 15) + "px";
        p.style.animationDuration = (Math.random() * 3 + 4) + "s";
        document.body.appendChild(p);
        setTimeout(() => p.remove(), 7000);
    }

    setInterval(createPetal, 300);

    gift.addEventListener('click', () => {
        lid.style.transform = 'translateY(-120px) rotate(-20deg)';
        lid.style.opacity = '0';
        sun.style.top = '70%'; 
        
        setTimeout(() => {
            container.style.display = 'none';
            message.style.display = 'block';
            document.body.style.background = "linear-gradient(to bottom, #2c3e50, #000000)"; 
            
            music.play().catch(e => console.log("Audio iniciado"));
            typeWriter(poema);
        }, 900);
    });

    document.addEventListener('touchstart', (e) => {
        const h = document.createElement('div');
        h.className = 'heart';
        h.innerText = '💜';
        h.style.left = e.touches[0].clientX + 'px';
        h.style.top = e.touches[0].clientY + 'px';
        document.body.appendChild(h);
        setTimeout(() => h.remove(), 1000);
    });
</script>
